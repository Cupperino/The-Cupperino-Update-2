		limit = {
			NOT = { check_variable = { ENG_trade_unions_support_shared > 0 } }
			NOT = { check_variable = { ENG_trade_unions_support_shared < 0 } }
		}
		set_variable = { var = ENG_trade_unions_support_shared value = 0 }
	}
	add_to_variable = { var = ENG_trade_unions_support_shared value = 10 }
	set_variable = { var = ENG_trade_unions_support_50_cost value = 50 }
	add_to_variable = { var = ENG_trade_unions_support_50_cost value = ENG_trade_unions_support_shared }
	set_variable = { var = ENG_trade_unions_support_75_cost value = 75 }
	add_to_variable = { var = ENG_trade_unions_support_75_cost value = ENG_trade_unions_support_shared }
	custom_effect_tooltip = ENG_request_trade_union_support_cost_level_up_tt
}

# The Shadow Scheme: count the civilian factories the UK has constructed this game (instantly built
# rewards are not constructions and do not count), work out how many military factories each country
# taking part is owed, and keep a target state picked for every country (scope: ENG)
ENG_shadow_scheme_track = {
	set_variable = { shadow_scheme_civs_built = total_constructed_civilian_factory }
	set_variable = { shadow_scheme_rounds = 0 }
	set_temp_variable = { shadow_scheme_next = shadow_scheme_civs_per_mil }
	while_loop_effect = {
		limit = {
			check_variable = { shadow_scheme_rounds < shadow_scheme_max_mils }
			check_variable = {
				var = shadow_scheme_civs_built
				value = shadow_scheme_next
				compare = greater_than_or_equals
			}
		}
		add_to_variable = { shadow_scheme_rounds = 1 }
		add_to_temp_variable = { shadow_scheme_next = shadow_scheme_civs_per_mil }
	}
	ENG_shadow_scheme_pick_target = yes
	CAN = { ENG_shadow_scheme_pick_target = yes }
	SAF = { ENG_shadow_scheme_pick_target = yes }
	AST = { ENG_shadow_scheme_pick_target = yes }
	NZL = { ENG_shadow_scheme_pick_target = yes }
	RAJ = { ENG_shadow_scheme_pick_target = yes }
}

# The Shadow Scheme: keep a core state with room for the factories flagged for the scoped country
ENG_shadow_scheme_pick_target = {
	if = {
		limit = {
			exists = yes
			NOT = {
				any_owned_state = {
					has_state_flag = ENG_shadow_scheme_target
					is_controlled_by = PREV
					is_core_of = PREV
				}
			}
		}
		every_owned_state = {
			limit = { has_state_flag = ENG_shadow_scheme_target }
			clr_state_flag = ENG_shadow_scheme_target
		}
		random_owned_controlled_state = {
			limit = {
				is_core_of = PREV
				free_building_slots = {
					building = arms_factory
					size > 4
					include_locked = yes
				}
			}
			set_state_flag = ENG_shadow_scheme_target
		}
	}
}

# The Military Industrialization Plan: count the civilian factories the UK has constructed this game, work out
# how many military factories it is owed, keep target states flagged and split the factories between them:
# full states get mil_plan_mils_per_state each, one partial state gets the remainder (mil_plan_last) (scope: ENG)
ENG_mil_plan_track = {
	set_variable = { mil_plan_civs_built = total_constructed_civilian_factory }
	# civs spawned by focuses count as built too
	add_to_variable = { mil_plan_civs_built = mil_plan_civs_spawned }
	set_variable = { mil_plan_mils = 0 }
	set_temp_variable = { mil_plan_next = mil_plan_civs_per_mil }
	while_loop_effect = {
		limit = {
			check_variable = { mil_plan_mils < mil_plan_max_mils }
			check_variable = {
				var = mil_plan_civs_built
				value = mil_plan_next
				compare = greater_than_or_equals
			}
		}
		add_to_variable = { mil_plan_mils = 1 }
		add_to_temp_variable = { mil_plan_next = mil_plan_civs_per_mil }
	}

	every_owned_state = {
		limit = {
			has_state_flag = ENG_mil_plan_target
			NOT = { ENG_is_mil_plan_candidate = yes }
		}
		clr_state_flag = ENG_mil_plan_target
	}
	set_temp_variable = { mil_plan_states_wanted = mil_plan_max_mils }
	divide_temp_variable = { mil_plan_states_wanted = mil_plan_mils_per_state }
	set_temp_variable = { mil_plan_states = 0 }
	every_owned_state = {
		limit = { has_state_flag = ENG_mil_plan_target }
		add_to_temp_variable = { mil_plan_states = 1 }
	}
	while_loop_effect = {
		limit = {
			check_variable = { mil_plan_states < mil_plan_states_wanted }
			any_owned_state = {
				ENG_is_mil_plan_candidate = yes
				NOT = { has_state_flag = ENG_mil_plan_target }
			}
		}
		random_owned_controlled_state = {
			limit = {
				ENG_is_mil_plan_candidate = yes
				NOT = { has_state_flag = ENG_mil_plan_target }
			}
			set_state_flag = ENG_mil_plan_target
		}
		add_to_temp_variable = { mil_plan_states = 1 }
	}

	every_owned_state = {
		limit = {
			OR = {
				has_state_flag = ENG_mil_plan_full
				has_state_flag = ENG_mil_plan_partial
			}
		}
		clr_state_flag = ENG_mil_plan_full
		clr_state_flag = ENG_mil_plan_partial
	}
	set_variable = { mil_plan_last = 0 }
	set_temp_variable = { mil_plan_remaining = mil_plan_mils }
	every_owned_state = {
		limit = { has_state_flag = ENG_mil_plan_target }
		if = {
			limit = {
				check_variable = {
					var = mil_plan_remaining
					value = ROOT.mil_plan_mils_per_state
					compare = greater_than_or_equals
				}
			}
			set_state_flag = ENG_mil_plan_full
			subtract_from_temp_variable = { mil_plan_remaining = ROOT.mil_plan_mils_per_state }
		}
		else_if = {
			limit = { check_variable = { mil_plan_remaining > 0 } }
			set_state_flag = ENG_mil_plan_partial
			set_variable = { ROOT.mil_plan_last = mil_plan_remaining }
			set_temp_variable = { mil_plan_remaining = 0 }
		}
	}
}

# State of the Empire: trade bonus by how much of Gibraltar and the Suez the UK holds (scope: ENG)

	if = {
		limit = {
			ENG_holds_gibraltar = yes
			ENG_holds_suez = yes
		}
		if = {
			limit = { NOT = { has_idea = ENG_state_of_the_empire_idea_1 } }
			ENG_clear_state_of_the_empire = yes
			add_ideas = ENG_state_of_the_empire_idea_1
		}
	}
	else_if = {
		limit = {
			OR = {
				ENG_holds_gibraltar = yes
				ENG_holds_suez = yes
			}
		}
		if = {
			limit = { NOT = { has_idea = ENG_state_of_the_empire_idea_2 } }
			ENG_clear_state_of_the_empire = yes
			add_ideas = ENG_state_of_the_empire_idea_2
		}
	}
	else_if = {
		limit = { NOT = { has_idea = ENG_state_of_the_empire_idea_3 } }
		ENG_clear_state_of_the_empire = yes
		add_ideas = ENG_state_of_the_empire_idea_3
	}
}
