# Containment: pacifists riot when we keep punishing Germany in a calm world
ENG_pacifist_riots_over_germany = {
	has_completed_focus = ENG_reign_in_the_reich
	GER = { exists = yes }
	NOT = { has_war_with = GER }
	check_variable = { threat < constant:ENG_pacifist_riots.germany_threat }
}

# Containment: pacifists riot when we keep the Italian embargo in a calm world
ENG_pacifist_riots_over_italy = {
	has_completed_focus = ENG_embargo_the_italians
	ITA = {
		exists = yes
		is_embargoed_by = ROOT
	}
	NOT = { has_war_with = ITA }
	check_variable = { threat < constant:ENG_pacifist_riots.italy_threat }
}

ENG_pacifist_riots_active = {
	OR = {
		ENG_pacifist_riots_over_germany = yes
		ENG_pacifist_riots_over_italy = yes
	}
}
